### API 测试用例设计

**文件**: `tests/extract/api/test_extract_api.py`

```python
class TestExtractAPI:
    """API 端点测试"""

    def test_upload_unsupported_file(self, client):
        """上传不支持的文件类型"""
        response = client.post('/api/extract/upload',
            data={'file': (io.BytesIO(b'xxx'), 'test.txt')})
        assert response.status_code == 400
        assert '不支持的文件类型' in response.json['error']

    def test_upload_award_success(self, client, sample_award):
        """上传奖状文件成功"""
        response = client.post('/api/extract/upload',
            data={'file': (open(sample_award, 'rb'), 'award.jpg')})
        assert response.status_code == 200
        data = response.json
        assert data['template_type'] == 'award'
        assert 'data' in data

    def test_get_result_success(self, client, extract_id):
        """获取抽取结果"""
        response = client.get(f'/api/extract/result/{extract_id}')
        assert response.status_code == 200
        data = response.json
        assert 'data' in data

    def test_batch_extract(self, client, sample_files):
        """批量抽取"""
        files = [('files', open(f, 'rb')) for f in sample_files]
        response = client.post('/api/extract/batch', data=files)
        assert response.status_code == 200
        results = response.json['results']
        assert len(results) == len(sample_files)

    def test_cache_hit(self, client, sample_award):
        """测试缓存命中"""
        # 第一次上传
        response1 = client.post('/api/extract/upload',
            data={'file': (open(sample_award, 'rb'), 'award.jpg')})
        data1 = response1.json

        # 第二次上传相同文件（应命中缓存）
        response2 = client.post('/api/extract/upload',
            data={'file': (open(sample_award, 'rb'), 'award.jpg')})
        data2 = response2.json

        assert data2['ocr_cache_hit'] == True or data2['llm_cache_hit'] == True
```

### 性能测试用例

```python
def test_performance_single_file(benchmark):
    """单文件性能测试"""
    result = benchmark(framework.extract, sample_award)
    assert result.status == ExtractStatus.SUCCESS

def test_performance_batch(batch_size=10):
    """批量处理性能测试"""
    start = time.time()
    for file in sample_files[:batch_size]:
        framework.extract(file)
    duration = time.time() - start
    assert duration < batch_size * 3  # 每个文件 < 3s
```
